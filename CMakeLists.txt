cmake_minimum_required(VERSION 3.24)
project(CS2FullReset VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows with MSVC is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(reset_core src/safe_fs.cpp src/backup.cpp src/accounts.cpp src/workflow.cpp)
target_include_directories(reset_core PUBLIC src)
enable_testing()
add_executable(reset_tests tests/tests.cpp)
target_link_libraries(reset_tests PRIVATE reset_core)
add_test(NAME fixtures COMMAND reset_tests)
add_executable(CS2FullReset src/main.cpp src/win_steam.cpp)
target_link_libraries(CS2FullReset PRIVATE reset_core user32 shell32 advapi32 ole32 oleaut32 uiautomationcore)
target_link_options(CS2FullReset PRIVATE /MANIFESTUAC:level='asInvoker' /OPT:REF /OPT:ICF)
add_test(NAME cli_help COMMAND CS2FullReset --help)
